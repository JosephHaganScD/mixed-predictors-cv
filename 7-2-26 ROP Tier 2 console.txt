> # rop_tier2_sensitivity_analysis.R
> #
> # Tier 2 Sensitivity Illustration: Minimal Fixed-Predictor Set Matched to a
> # Simulated Arm C Condition, Evaluated Across Three Longitudinal Predictor
> # Variants and Two Learners (Ridge, XGBoost)
> #
> # Fixed set: BWGT + GA only (p_F = 2, continuous only; avoids the
> #   categorical/continuous type mismatch present in the Tier 1 8-predictor
> #   fixed set). BWGT + GA alone uniquely identify all 100 subjects (Section
> #   3.6), so this isolates identity-mediated leakage in a minimal,
> #   EPV-favorable model.
> #
> # Longitudinal variants:
> #   V1: Avg_FiO2 + Severe_Hypoxemia__            (p_L = 2; matches simulated
> #       Arm C condition (p_F, p_L) = (2, 2))
> #   V2: Avg_FiO2 + Avg_SpO2                      (p_L = 2; matches (2, 2))
> #   V3: Avg_FiO2 + Avg_SpO2 + Severe_Hypoxemia__ (p_L = 3; NOT a simulated
> #       Arm C combination. Included as an additional robustness check only.
> #       Do not present as a matched-condition comparator if used in the
> #       manuscript.)
> #
> # Learners:
> #   Ridge logistic regression, fixed lambda = log(1 + 1/n) / n (n = subjects),
> #     matching Tier 1 and the simulation.
> #   XGBoost, max_depth = 4, eta = 0.1, nrounds = 50, matching the simulation
> #     (Section 2.3). Fit at the row level; subject-level predicted
> #     probability obtained by averaging within-subject row predictions.
> #
> # CV strategies: naive 10-fold, subject-level 10-fold, LOCO (all three, for
> #   both learners, consistent with Tier 1).
> # Outcomes: any_rop (primary, 48 events), severe_rop (secondary, 15 events).
> #
> # CAUTION: XGBoost LOCO for severe_rop is expected to be unstable (15 events,
> # 100 subjects); point estimates and DeLong CIs may be wide. Computed
> # regardless; frame the interpretation accordingly if reported.
> #
> # Author: Joseph L. Hagan, ScD, MSPH
> # Section of Neonatology, Baylor College of Medicine

> fixed_lambda <- function(n_subj) log(1 + 1 / n_subj) / n_subj

> # Ridge logistic regression. lam supplied by caller.
> fit_ridge <- function(X_train, y_train, X_test, lam, n_subj = NULL) {
+   fit <- tryCatch(
+     glmnet(X_train, y_train, family = "binomial", alpha = 0,
+            lambda = lam, standardize = TRUE),
+     error = function(e) {
+       glmnet(X_train, y_train, family = "binomial", alpha = 0,
+              lambda = max(lam, 0.01), standardize = TRUE)
+     }
+   )
+   as.numeric(predict(fit, newx = X_test, s = fit$lambda[1L], type = "response"))
+ }
> 
> # XGBoost, hyperparameters fixed to match simulation Section 2.3.
> # lam argument accepted but ignored, so fit_ridge and fit_xgb share a
> # signature and can both be passed as fit_fn to the CV functions below.
> fit_xgb <- function(X_train, y_train, X_test, lam = NULL, n_subj) {
+   mcw <- max(1L, floor(n_subj / 20))
+   dtr <- xgboost::xgb.DMatrix(data = X_train, label = y_train)
+   fit <- xgboost::xgb.train(
+     params  = list(objective = "binary:logistic", max_depth = 4, eta = 0.1,
+                     min_child_weight = mcw, nthread = 1),
+     data    = dtr,
+     nrounds = 50,
+     verbose = 0
+   )
+   as.numeric(predict(fit, X_test))
+ }

> row_to_subj_auroc <- function(case_ids, row_preds, row_y) {
+   subj_p <- tapply(row_preds, case_ids, mean)
+   subj_y <- tapply(row_y,    case_ids, function(x) x[1L])
+   subj_y <- subj_y[names(subj_p)]
+   if (length(unique(subj_y)) < 2L) return(NA_real_)
+   as.numeric(pROC::auc(pROC::roc(as.numeric(subj_y), as.numeric(subj_p),

+ }

> cv_naive <- function(X, y, case_ids, fit_fn, k = K_FOLDS, seed = 1L) {
+   set.seed(seed)
+   n       <- nrow(X)
+   lam     <- fixed_lambda(length(unique(case_ids)))
+   fold_id <- sample(rep(seq_len(k), length.out = n))
+   preds   <- rep(NA_real_, n)
+   for (f in seq_len(k)) {
+     tr <- fold_id != f; te <- fold_id == f
+     if (length(unique(y[tr])) < 2L) next
+     n_subj_f  <- length(unique(case_ids[tr]))
+     preds[te] <- fit_fn(X[tr, , drop = FALSE], y[tr], X[te, , drop = FALSE],
+                          lam, n_subj_f)
+   }
+   valid <- !is.na(preds)
+   row_to_subj_auroc(case_ids[valid], preds[valid], y[valid])
+ }
> 
> cv_subject <- function(X, y, case_ids, fit_fn, k = K_FOLDS, seed = 1L) {
+   set.seed(seed)
+   lam      <- fixed_lambda(length(unique(case_ids)))
+   subj_ids <- unique(case_ids)
+   n_s      <- length(subj_ids)
+   k_act    <- min(k, n_s)
+   fold_map <- setNames(rep(seq_len(k_act), length.out = n_s),
+                        subj_ids[sample(n_s)])
+   obs_fold <- fold_map[as.character(case_ids)]
+   preds    <- rep(NA_real_, nrow(X))
+   for (f in seq_len(k_act)) {
+     te <- which(obs_fold == f); tr <- which(obs_fold != f)
+     if (length(te) == 0L || length(unique(y[tr])) < 2L) next
+     n_subj_f  <- length(unique(case_ids[tr]))
+     preds[te] <- fit_fn(X[tr, , drop = FALSE], y[tr], X[te, , drop = FALSE],
+                          lam, n_subj_f)
+   }
+   valid <- !is.na(preds)
+   row_to_subj_auroc(case_ids[valid], preds[valid], y[valid])
+ }
> 
> cv_loco <- function(X, y, case_ids, fit_fn) {
+   lam      <- fixed_lambda(length(unique(case_ids)))
+   subj_ids <- unique(case_ids)
+   subj_p   <- setNames(numeric(length(subj_ids)), subj_ids)
+   for (id in subj_ids) {
+     te <- which(case_ids == id); tr <- which(case_ids != id)
+     if (length(unique(y[tr])) < 2L) { subj_p[as.character(id)] <- 0.5; next }
+     n_subj_f <- length(unique(case_ids[tr]))
+     subj_p[as.character(id)] <- mean(
+       fit_fn(X[tr, , drop = FALSE], y[tr], X[te, , drop = FALSE],
+              lam, n_subj_f))
+   }
+   subj_y <- tapply(y, case_ids, function(x) x[1L])
+   subj_y <- subj_y[names(subj_p)]
+   if (length(unique(subj_y)) < 2L) return(list(auc = NA_real_, roc = NULL))
+   roc_obj <- pROC::roc(as.numeric(subj_y), as.numeric(subj_p), quiet = TRUE)
+   list(auc = as.numeric(pROC::auc(roc_obj)), roc = roc_obj)
+ }

> # =============================================================================
> # 4. RUN ONE (predictor set x learner x outcome) CONFIGURATION
> # =============================================================================
> 
> run_config <- function(dat, outcome_var, pred_cols, config_label, fit_fn,
+                        n_reps = N_REPS, k = K_FOLDS) {
+ 
+   cat(sprintf("\n  --- %s ---\n", config_label))
+ 
+   X        <- as.matrix(dat[, pred_cols, drop = FALSE])
+   y        <- dat[[outcome_var]]
+   case_ids <- dat$Case_ID
+ 
+   cat("    Naive CV...")
+   naive_aucs <- sapply(seq_len(n_reps), function(r)
+     cv_naive(X, y, case_ids, fit_fn, k = k, seed = r))
+   cat(" done\n")
+ 
+   cat("    Subject CV...")
+   subj_aucs <- sapply(seq_len(n_reps), function(r)
+     cv_subject(X, y, case_ids, fit_fn, k = k, seed = r))
+   cat(" done\n")
+ 
+   cat("    LOCO CV...")
+   loco_res <- cv_loco(X, y, case_ids, fit_fn)
+   loco_auc <- loco_res$auc
+   loco_ci  <- if (!is.null(loco_res$roc))
+                 as.numeric(pROC::ci.auc(loco_res$roc, method = "delong"))
+               else c(NA, NA, NA)
+   cat(" done\n")
+ 
+   naive_opt <- naive_aucs - loco_auc
+   subj_dev  <- subj_aucs  - loco_auc
+ 
+   list(
+     config         = config_label,
+     naive_aucs     = naive_aucs,
+     subj_aucs      = subj_aucs,
+     loco_auc       = loco_auc,
+     loco_ci        = loco_ci,
+     naive_opt      = naive_opt,
+     subj_dev       = subj_dev,
+     mean_naive     = mean(naive_aucs, na.rm = TRUE),
+     mean_subj      = mean(subj_aucs,  na.rm = TRUE),
+     mean_naive_opt = mean(naive_opt,  na.rm = TRUE),
+     mean_subj_dev  = mean(subj_dev,   na.rm = TRUE),
+     pi_naive       = quantile(naive_aucs, c(0.025, 0.975), na.rm = TRUE),
+     pi_subj        = quantile(subj_aucs,  c(0.025, 0.975), na.rm = TRUE),
+     pi_naive_opt   = quantile(naive_opt,  c(0.025, 0.975), na.rm = TRUE),
+     pi_subj_dev    = quantile(subj_dev,   c(0.025, 0.975), na.rm = TRUE)
+   )
+ }
